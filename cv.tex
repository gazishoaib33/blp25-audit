\section{Research}

\textbf{Preprints and Manuscripts}
\begin{itemize}
\item \textbf{Gazi, S.} (2026). ``Indistinguishable at the Top: A Statistical
      Audit of the BLP-2025 Bangla Hate Speech Shared Task.''
      \textit{Preprint}, Zenodo.
      \href{https://doi.org/10.5281/zenodo.22472234}{doi:10.5281/zenodo.22472234}\\
      Code: \href{https://github.com/gazishoaib33/blp25-audit}{github.com/gazishoaib33/blp25-audit}
      \begin{itemize}
      \item Showed the top five systems in all three subtasks are statistically
            indistinguishable, and trained ten systems to measure the pairwise
            discordance the argument depends on across 45 system pairs.
      \item Estimated the annotation ceiling from reported inter-annotator
            agreement and \emph{rejected} the hypothesis that label noise
            explains the observed performance plateau.
      \item Demonstrated that the task's micro-F1 metric cannot resolve its
            rarest class: at 29 of 10,200 test items, the metric's maximum
            response falls below the sampling noise floor.
      \end{itemize}
\end{itemize}
